```python
# watermeter/auth.py
"""Authentication manager for watermeter dashboard."""

import json
import secrets
import time
from pathlib import Path

import bcrypt
import jwt


class AuthManager:
    """Manages admin authentication — password hashing, JWT tokens, auth.json persistence."""

    def __init__(self, auth_file: Path, jwt_secret: str | None = None):
        self.auth_file = auth_file
        self._jwt_secret = jwt_secret
        self._auth_data: dict | None = None
        self._load()

    def _load(self):
        """Load auth.json if it exists."""
        if self.auth_file.exists():
            self._auth_data = json.loads(self.auth_file.read_text())
            if not self._jwt_secret:
                self._jwt_secret = self._auth_data.get("jwt_secret")

    def _save(self):
        """Write auth.json atomically."""
        self.auth_file.parent.mkdir(parents=True, exist_ok=True)
        tmp = self.auth_file.with_suffix(".tmp")
        tmp.write_text(json.dumps(self._auth_data, indent=2))
        tmp.rename(self.auth_file)

    # -- Password hashing --

    def hash_password(self, password: str) -> str:
        return bcrypt.hashpw(password.encode(), bcrypt.gensalt()).decode()

    def verify_password(self, password: str, hashed: str) -> bool:
        return bcrypt.checkpw(password.encode(), hashed.encode())
```

```python
    # -- Setup --

    def is_configured(self) -> bool:
        """True if a password has been set (auth.json exists with hash)."""
        return self._auth_data is not None and "admin_password_hash" in self._auth_data

    def setup(self, password: str):
        """First-time setup: set admin password, generate JWT secret, save auth.json."""
        if not self._jwt_secret:
            self._jwt_secret = secrets.token_hex(32)
        self._auth_data = {
            "jwt_secret": self._jwt_secret,
            "admin_password_hash": self.hash_password(password),
            "created_at": time.strftime("%Y-%m-%dT%H:%M:%S%z"),
        }
        self._save()

    def check_password(self, password: str) -> bool:
        """Verify password against stored hash."""
        if not self.is_configured():
            return False
        return self.verify_password(password, self._auth_data["admin_password_hash"])

    def change_password(self, new_password: str):
        """Change the admin password. Preserves JWT secret (existing tokens stay valid)."""
        if not self.is_configured():
            raise RuntimeError("Auth not configured — use setup() first")
        self._auth_data["admin_password_hash"] = self.hash_password(new_password)
        self._save()

    # -- JWT --

    def create_token(self, lifetime: int = 86400) -> str:
        """Create a JWT token for the admin user."""
        now = time.time()
        payload = {
            "sub": "admin",
            "iat": int(now),
            "exp": int(now + lifetime),
        }
        return jwt.encode(payload, self._jwt_secret, algorithm="HS256")

    def verify_token(self, token: str) -> dict | None:
        """Verify and decode a JWT token. Returns payload dict or None if invalid/expired."""
        try:
            return jwt.decode(token, self._jwt_secret, algorithms=["HS256"])
        except (jwt.InvalidTokenError, jwt.ExpiredSignatureError):
            return None

    def should_refresh(self, payload: dict, threshold: int = 1800) -> bool:
        """True if the token should be refreshed (expires within threshold seconds)."""
        return payload["exp"] - time.time() < threshold
```

```python
# watermeter/auth_middleware.py
"""Authentication middleware for watermeter dashboard."""

from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request
from starlette.responses import JSONResponse, RedirectResponse


# Exact paths that are public (no auth required)
PUBLIC_PATHS = frozenset({
    "/",
    "/login",
    "/setup",
    "/api/health",
    "/api/status",
    "/api/status/html",
    "/api/auth/login",
    "/api/auth/setup",
    "/api/auth/logout",
    "/docs",
    "/redoc",
    "/openapi.json",
})

# Path prefixes that are public
PUBLIC_PREFIXES = ("/static/",)


class AuthMiddleware(BaseHTTPMiddleware):

    def __init__(self, app, auth_manager, auth_config: dict):
        super().__init__(app)
        self.auth_manager = auth_manager
        self.cookie_name = auth_config.get("cookie_name", "watermeter_token")
        self.refresh_threshold = auth_config.get("refresh_threshold", 1800)
        self.token_lifetime = auth_config.get("token_lifetime", 86400)
        self.cookie_secure = auth_config.get("cookie_secure", False)

    async def dispatch(self, request: Request, call_next):
        path = request.url.path

        # Public routes — pass through
        if path in PUBLIC_PATHS or any(path.startswith(p) for p in PUBLIC_PREFIXES):
            return await call_next(request)

        is_api = path.startswith("/api/")
        is_htmx = request.headers.get("HX-Request") == "true"

        # Setup mode — no password set yet
        if not self.auth_manager.is_configured():
            if is_api:
                resp = JSONResponse(
                    status_code=401,
                    content={"detail": "Auth not configured — visit /setup"},
                )
                if is_htmx:
                    resp.headers["HX-Redirect"] = "/setup"
                return resp
            return RedirectResponse(url="/setup", status_code=307)

        # Extract token from cookie or Bearer header
        token = request.cookies.get(self.cookie_name)
        if not token:
            auth_header = request.headers.get("Authorization", "")
            if auth_header.startswith("Bearer "):
                token = auth_header[7:]

        # No token
        if not token:
            return self._unauthorized(request, path, is_api, is_htmx)

        # Verify token
        payload = self.auth_manager.verify_token(token)
        if not payload:
            return self._unauthorized(request, path, is_api, is_htmx)

        # Proceed with request
        response = await call_next(request)

        # Auto-refresh token if near expiry
        if self.auth_manager.should_refresh(payload, self.refresh_threshold):
            new_token = self.auth_manager.create_token(lifetime=self.token_lifetime)
            response.set_cookie(
                key=self.cookie_name,
                value=new_token,
                httponly=True,
                samesite="lax",
                secure=self.cookie_secure,
            )

        return response

    def _unauthorized(self, request, path, is_api, is_htmx):
        if is_api:
            resp = JSONResponse(
                status_code=401,
                content={"detail": "Not authenticated"},
            )
            if is_htmx:
                resp.headers["HX-Redirect"] = "/login"
            return resp
        # Redirect to login with next= parameter
        next_url = path
        if request.url.query:
            next_url += f"?{request.url.query}"
        return RedirectResponse(url=f"/login?next={next_url}", status_code=307)
```

```python
# watermeter/routes/auth.py
"""Authentication routes — login, logout, setup, password change."""

import logging
from fastapi import APIRouter, Form, Request
from fastapi.responses import JSONResponse, RedirectResponse

logger = logging.getLogger(__name__)


def create_auth_router(auth_manager, auth_config: dict) -> APIRouter:
    router = APIRouter(tags=["auth"])
    cookie_name = auth_config.get("cookie_name", "watermeter_token")
    token_lifetime = auth_config.get("token_lifetime", 86400)
    cookie_secure = auth_config.get("cookie_secure", False)

    def _safe_next_url(next_url: str | None) -> str:
        """Validate next URL to prevent open redirect."""
        if not next_url or not next_url.startswith("/") or next_url.startswith("//"):
            return "/"
        return next_url

    @router.post("/api/auth/setup")
    async def setup_password(
        password: str = Form(""),
        password_confirm: str = Form(""),
    ):
        if auth_manager.is_configured():
            return JSONResponse(status_code=400, content={"detail": "Already configured"})
        if not password:
            return JSONResponse(status_code=400, content={"detail": "Password cannot be empty"})
        if password != password_confirm:
            return JSONResponse(status_code=400, content={"detail": "Passwords do not match"})
        auth_manager.setup(password)
        logger.info("Admin password configured via setup")
        return JSONResponse(content={"detail": "Admin password set. Please log in."})

    @router.post("/api/auth/login")
    async def login(request: Request):
        content_type = request.headers.get("content-type", "")

        if "application/json" in content_type:
            body = await request.json()
            password = body.get("password", "")
            next_url = body.get("next")
        else:
            form = await request.form()
            password = form.get("password", "")
            next_url = form.get("next")

        if not auth_manager.check_password(password):
            if "application/json" in content_type:
                return JSONResponse(status_code=401, content={"detail": "Invalid credentials"})
            return JSONResponse(status_code=401, content={"detail": "Invalid credentials"})

        token = auth_manager.create_token(lifetime=token_lifetime)

        if "application/json" in content_type:
            return JSONResponse(content={"token": token})

        # Browser form login — redirect with cookie
        resp = RedirectResponse(url=_safe_next_url(next_url), status_code=303)
        resp.set_cookie(
            key=cookie_name,
            value=token,
            httponly=True,
            samesite="lax",
            secure=cookie_secure,
        )
        return resp

    @router.post("/api/auth/logout")
    async def logout():
        resp = RedirectResponse(url="/login", status_code=303)
        resp.delete_cookie(key=cookie_name)
        return resp

    @router.post("/api/auth/change-password")
    async def change_password(request: Request):
        body = await request.json()
        current = body.get("current_password", "")
        new = body.get("new_password", "")

        if not auth_manager.check_password(current):
            return JSONResponse(status_code=401, content={"detail": "Current password incorrect"})
        if not new:
            return JSONResponse(status_code=400, content={"detail": "New password cannot be empty"})

        auth_manager.change_password(new)
        logger.info("Admin password changed")
        return JSONResponse(content={"detail": "Password changed"})

    return router
```
